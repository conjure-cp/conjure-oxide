Model before rewriting:

find x: int(5..20)
find y: int(0..5)
find z: int(0..6)

such that

(x / (y / z)) = 10

--

(y / z)
   ~~> div_to_bubble ([("Bubble", 6000)])
{SafeDiv(y, z) @ z != 0}

--

(x / {SafeDiv(y, z) @ z != 0})
   ~~> bubble_up ([("Bubble", 8800)])
{(x / SafeDiv(y, z)) @ z != 0}

--

{(x / SafeDiv(y, z)) @ z != 0} = 10
   ~~> bubble_up ([("Bubble", 8800)])
{(x / SafeDiv(y, z)) = 10 @ z != 0}

--

{(x / SafeDiv(y, z)) = 10 @ z != 0}
   ~~> expand_bubble ([("Bubble", 8900)])
and([(x / SafeDiv(y, z)) = 10,z != 0;int(1..)])

--

(x / SafeDiv(y, z))
   ~~> div_to_bubble ([("Bubble", 6000)])
{SafeDiv(x, SafeDiv(y, z)) @ SafeDiv(y, z) != 0}

--

{SafeDiv(x, SafeDiv(y, z)) @ SafeDiv(y, z) != 0} = 10
   ~~> bubble_up ([("Bubble", 8800)])
{SafeDiv(x, SafeDiv(y, z)) = 10 @ SafeDiv(y, z) != 0}

--

{SafeDiv(x, SafeDiv(y, z)) = 10 @ SafeDiv(y, z) != 0}
   ~~> expand_bubble ([("Bubble", 8900)])
and([SafeDiv(x, SafeDiv(y, z)) = 10,SafeDiv(y, z) != 0;int(1..)])

--

and([and([SafeDiv(x, SafeDiv(y, z)) = 10,SafeDiv(y, z) != 0;int(1..)]),z != 0;int(1..)])
   ~~> normalise_associative_commutative ([("Base", 8900)])
and([SafeDiv(x, SafeDiv(y, z)) = 10,SafeDiv(y, z) != 0,z != 0;int(1..)])

--

SafeDiv(y, z) != 0
   ~~> flatten_generic ([("Minion", 4200)])
__0 != 0
new variables:
  find __0: int(0..5)
new constraints:
  __0 =aux SafeDiv(y, z)

--

__0 =aux SafeDiv(y, z)
   ~~> introduce_diveq ([("Minion", 4200)])
DivEq(y, z, __0)

--

SafeDiv(x, SafeDiv(y, z))
   ~~> flatten_generic ([("Minion", 4200)])
SafeDiv(x, __1)
new variables:
  find __1: int(0..5)
new constraints:
  __1 =aux SafeDiv(y, z)

--

__1 =aux SafeDiv(y, z)
   ~~> introduce_diveq ([("Minion", 4200)])
DivEq(y, z, __1)

--

SafeDiv(x, __1) = 10
   ~~> introduce_diveq ([("Minion", 4200)])
DivEq(x, __1, 10)

--

Final model:

find x: int(5..20)
find y: int(0..5)
find z: int(0..6)
find __0: int(0..5)
find __1: int(0..5)

such that

and([DivEq(x, __1, 10),__0 != 0,z != 0;int(1..)]),
DivEq(y, z, __0),
DivEq(y, z, __1)
