such that

b != 0,

quotient = (a / b)

quotient = {SafeDiv(a, b) @ b != 0}
   ~~> bubble_up ([("Bubble", 8800)])
{quotient = SafeDiv(a, b) @ b != 0}

   ~~> expand_bubble ([("Bubble", 8900)])
