Model before rewriting:

find x: int(3..3)

such that

(x / 2) = 1,
(x / -2) = -1,
(-(x) / 2) = -1,
(-(x) / -2) = 1

--

(x / 2)
   ~~> div_to_bubble ([("Bubble", 6000)])
{SafeDiv(x, 2) @ 2 != 0}

--

(x / -2)
   ~~> div_to_bubble ([("Bubble", 6000)])
{SafeDiv(x, -2) @ -2 != 0}

--

(-(x) / 2)
   ~~> div_to_bubble ([("Bubble", 6000)])
{SafeDiv(-(x), 2) @ 2 != 0}

--

(-(x) / -2)
   ~~> div_to_bubble ([("Bubble", 6000)])
{SafeDiv(-(x), -2) @ -2 != 0}

--

SafeDiv(x, 2) = 1
   ~~> introduce_diveq ([("Minion", 4200)])

SafeDiv(x, -2) = -1
   ~~> introduce_diveq ([("Minion", 4200)])
DivEq(x, -2, -1)

   ~~> flatten_generic ([("Minion", 4200)])

  findAux __0: int(-3..-3)

  __0 =aux -(x)

   ~~> introduce_diveq ([("Minion", 4200)])
DivEq(__0, 2, -1)

--

SafeDiv(-(x), -2)
   ~~> flatten_generic ([("Minion", 4200)])
SafeDiv(__1, -2)

  findAux __1: int(-3..-3)
